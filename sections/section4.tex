\section{Modellazione Cinematica e Controllo}
\begin{frame}{Modelli e Notazione Usata}
    Confrontati due modelli cinematici:
    \begin{itemize}
        \item \textbf{Biciclo Sterzante}: Preso dalla letteratura\footnote{Tourajizadeh et al., ``Modeling and Optimal Control of 4 Wheel Steering Vehicle Using LQR'', 2020;}. Approssima un veicolo con quattro ruote sterzanti.
        \item \textbf{Doppia Ruota Sterzante}: proposta basata su vincoli non-olonomi, senza approssimazioni geometriche.
    \end{itemize}
    \begin{columns}

        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \item \textbf{Stato}: $q=\begin{bmatrix}
                              x & y & \theta
                          \end{bmatrix}^\top$,
                \item \textbf{Dimensioni}: $L,W$, $L>W$,
                \item \textbf{Posizioni delle Ruote}:
                      \vspace{-0.2cm}
                      \begin{align*}
                          \mathbf{p}_f^b & =\begin{bmatrix}
                                                L/2 & W/2
                                            \end{bmatrix}^\top  \\
                          \mathbf{p}_r^b & =-\begin{bmatrix}
                                                 L/2 & W/2
                                             \end{bmatrix}^\top
                      \end{align*}
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \item \textbf{Input di Controllo}: $\mathbf{u}=\begin{bmatrix}
                              \omega_f & \omega_r & \delta_f & \delta_r
                          \end{bmatrix}^\top$,
                \item \textbf{Velocità Angolare Ruote}: $\omega_f$, $\omega_r$,
                \item \textbf{Angoli di Sterzo}: $\delta_f$, $\delta_r$,
                \item \textbf{Raggio Ruote}: $r$.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
\begin{frame}
    \frametitle{Ancora sulla notazione}
    \begin{itemize}
        \item \textbf{Vel. Centro Robot (body frame)}: $\dot{x}_c^b, \dot{y}_c^b$
        \item \textbf{Assunzione}: simmetria geometrica
              \begin{itemize}
                  \item Centro geometrico coincide con centro di massa
                  \item Distanza tra centro e asse ruote: $l_f = l_r = L/2\Rightarrow l_f+l_r=L$
              \end{itemize}
        \item \textbf{Convenzioni}: angoli positivi in senso antiorario
              \begin{itemize}
                  \item \textbf{$\theta$} zero quando asse longitudinale parallelo a asse $x$ globale
                  \item \textbf{Sterzi} zero quando ruote parallele all'asse longitudinale
              \end{itemize}
        \item \textbf{Velocità angolari positive}: rotazione in avanti (avanzamento)
    \end{itemize}
\end{frame}
\begin{frame}
    \frametitle{Modello a Biciclo Sterzante}
    \textbf{Semplificazione}: ruote sullo stesso asse $\rightarrow$ ruota virtuale unica per asse

    \vspace{0.3cm}
    \begin{columns}
        \begin{column}{0.5\textwidth}

            \textbf{Equazioni Cinematiche}
            \begin{equation}\label{eq:bicycle}
                \begin{cases}
                    \dot{x}      = v \cos(\theta+\beta) \\
                    \dot{y}      = v \sin(\theta+\beta) \\
                    \dot{\theta} = \frac{v\cos(\beta)}{l_f+l_r}\left(\tan(\delta_f)-\tan(\delta_r)\right)
                \end{cases}
            \end{equation}

            dove:
            \begin{align}
                v     & =  \frac{r(\omega_f \cos\delta_f + \omega_r \cos\delta_r)}{2\cos(\beta)}                                                    \\
                \beta & =  \tan^{-1}\left(\frac{l_f \tan(\delta_r) + l_r \tan(\delta_f)}{l_f + l_r}\right) \quad \text{(side-slip)} \label{eq:beta}
            \end{align}

            \vspace{0.2cm}
        \end{column}
        \begin{column}{0.4\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=0.9\textwidth]{img/biciclo.png}
                \caption{Modello a Biciclo Sterzante}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}
\begin{frame}
    \frametitle{Modello a Biciclo Sterzante: Osservazioni}
    \begin{block}{Vantaggi}
        \begin{itemize}
            \item Testato in letteratura.
            \item Modello dinamico studiato (controllabile con LQR).
        \end{itemize}
    \end{block}
    \begin{alertblock}{Svantaggi: Singolarità}
        \begin{itemize}
            \item $\beta \rightarrow \infty$ se $\delta_f$ o $\delta_r \rightarrow \pi/2$ (tangente infinita)
            \item \textbf{Crab/Spin impossibile}: se $v=0$ allora $\omega_f = \omega_r = 0 \Rightarrow \dot{\theta}=0$
        \end{itemize}
    \end{alertblock}
\end{frame}

\begin{frame}
    \frametitle{Modello a Doppia Ruota Sterzante}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Soluzione finale} nel body frame:

            \begin{equation}
                \begin{bmatrix}
                    \dot{x}_c^b \\ \dot{y}_c^b \\ \dot{\theta}
                \end{bmatrix}
                = \frac{r}{2}
                \begin{bmatrix}
                    \cos(\delta_f) & \cos(\delta_r) \\
                    \sin(\delta_f) & \sin(\delta_r) \\
                    a_{13}         & a_{23}
                \end{bmatrix}
                \begin{bmatrix}
                    \omega_f \\ \omega_r
                \end{bmatrix}
            \end{equation}

            dove i coefficienti $a_{13}$, $a_{23}$ dipendono da $\delta_f$, $\delta_r$, $L$, $W$:
            \begin{align}
                a_{13} & = \frac{2(L\sin(\delta_f) - W\cos(\delta_f))}{L^2 + W^2} \\
                a_{23} & = \frac{2(W\cos(\delta_r) - L\sin(\delta_r))}{L^2 + W^2}
            \end{align}
        \end{column}
        \begin{column}{0.4\textwidth}
            \begin{figure}[htbp]
                \centering
                \includegraphics[width=0.8\textwidth]{2wd_steering.png}
                \caption{Ruote motrici in verde, passive in viola.}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}
\begin{frame}
    \frametitle{Modello a Doppia Ruota Sterzante: Osservazioni}
    \begin{block}{Vantaggi}
        \begin{itemize}
            \item \textbf{Nessuna singolarità cinematica} per qualsiasi $\delta_f, \delta_r$.
            \item Gestisce \textbf{crab} ($\dot{x}_c=0$, $\dot{y}_c\neq0$) e \textbf{spin} ($\dot{x}_c=\dot{y}_c=0$, $\dot{\theta}\neq0$)
        \end{itemize}
    \end{block}
    \begin{alertblock}{Svantaggi}
        \begin{itemize}
            \item No modellazione dinamica (forze, inerzie).
            \item No modellazione ruote libere.
            \item Soluzione ai minimi quadrati: possibilità di violare vincoli non-olonomi.
        \end{itemize}
    \end{alertblock}
\end{frame}


\begin{frame}
    \frametitle{Setup Simulazioni}
    \textbf{Parametri Robot e Attuatori}
    \begin{itemize}
        \item Dimensioni: $L = 2$ m, $W = 1$ m, $d = 0.198\Rightarrow r= 0.099$ m
        \item Dinamica attuatori: modello 1$^\circ$ ordine $P(s) = \frac{K}{\tau s + 1}$
              \begin{itemize}
                  \item Trazione: $K_{\omega} = 1$, $\tau_\omega = 1$ s, $\omega_{max} = 10$ rad/s
                  \item Sterzo: $K_{\delta} = 1$, $\tau_\delta = 2$ s, $\delta_{max} = \pm 120^\circ$
              \end{itemize}
    \end{itemize}

    \vspace{0.3cm}

    \textbf{Traiettorie di Test}
    \begin{enumerate}
        \item \textbf{Straight}: linea retta verso Est ($v_{ref} = 0.5$ m/s, $T = 5$ s)
        \item \textbf{Arc}: arco circolare ($R = 3$ m, $v_{ref} = 0.6$ m/s, $\omega = 0.2$ rad/s, $T = 8$ s)
        \item \textbf{Crab}: traslazione laterale Nord ($v_{ref} = 0.4$ m/s, $T = 5$ s)
        \item \textbf{Spin}: rotazione sul posto ($\omega_{ref} = \pi/6$ rad/s, $T = 6$ s)
    \end{enumerate}

\begin{frame}
    \frametitle{Dettagli Implementazione PID}
    \textbf{Legge di controllo}:
    \begin{align}
        e^b                           & = R^{-1}(\theta) \begin{bmatrix} x_d - x \\ y_d - y \\ \theta_d - \theta \end{bmatrix} \quad \text{(errore in body frame)} \\
        \dot{q}_{i_{\text{PID}}}^b(t) & = K_{p,i} e_i^b + K_{i,i} \int e_i^b \, d\tau + K_{d,i} \frac{d e_i^b}{dt}, \quad i \in \{x,y,\theta\}                     \\
        \dot{q}_{\text{cmd}}          & = \dot{q}^d+R(\theta)\dot{q}_{i_{\text{PID}}}^b                                                                            \\
        \mathbf{u}                    & = \text{IK}(\dot{q}_{\text{cmd}})
    \end{align}
\end{frame}

\begin{frame}
    \frametitle{Dettagli Implementazione PID}
    \textbf{Guadagni utilizzati}:
    \begin{table}
        \centering
        \small
        \begin{tabular}{cccccc}
            \toprule
            \textbf{Comp.} & $K_p$ & $K_i$ & $K_d$ & $I_{max}$ & $N$ \\
            \midrule
            $x, y$         & 0.3   & 0.05  & 0.1   & 2         & 20  \\
            $\theta$       & 0.2   & 0.02  & 0.05  & 1         & 20  \\
            \bottomrule
        \end{tabular}
    \end{table}

    \vspace{0.3cm}

    \textbf{Accorgimenti implementativi}:
    \begin{itemize}
        \item \textbf{Derivata filtrata}: $D(s) = \frac{K_d s}{s/N + 1}$ (riduce rumore)
        \item \textbf{Anti-windup}: saturazione integrale per evitare wind-up
        \item \textbf{Scaling proporzionale}: mantiene rapporto tra $u_x, u_y$ sotto $v_{max}$
        \item \textbf{Saturazione angolare}: $|u_\theta| \leq \omega_{max}$
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Conclusioni: Confronto Modelli Cinematici}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Modello Biciclo}
        \begin{itemize}
            \item[+] Testato in letteratura
            \item[+] Estensione dinamica disponibile
            \item[-] \textbf{Singolarità}: Crab e Spin impossibili
            \item[-] $\beta \to \infty$ per $\delta \to \pi/2$
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Modello Doppio Sterzo}
        \begin{itemize}
            \item[+] \textbf{Nessuna singolarità}
            \item[+] Supporta tutte le manovre
            \item[+] Prestazioni migliori con PID
            \item[!] Pseudoinversa può violare vincoli
        \end{itemize}
    \end{columns}
