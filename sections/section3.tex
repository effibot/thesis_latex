% Slide 2: Dimensionamento Potenza
\begin{frame}
    \frametitle{Dimensionamento Potenza}
    Calcolo forza di trazione necessaria ($m=1000$ kg):
    \begin{itemize}
        \item \textbf{Piano}: $F_{piano} = \mu \cdot m \cdot g \approx 98$ N ($\mu=0.01$).
        \item \textbf{Pendenza (10\%)}: $F_{salita} = m \cdot g \cdot (\sin\theta + \mu\cos\theta) \approx 1078$ N.
    \end{itemize}

    \vspace{0.5cm}
    \begin{block}{Potenza Richiesta}
        $$ P_{req} = F_{salita} \cdot v_{max} \approx 1078 \text{ N} \cdot 1 \text{ m/s} \approx \text{\textbf{1.1 kW}} $$
    \end{block}
\end{frame}

% Slide 3: Selezione Batteria
\begin{frame}
    \frametitle{Selezione Batteria}
    \begin{columns}
        \column{0.6\textwidth}
        \textbf{FlashBattery LiFePO4} (Litio-Ferro-Fosfato)
        \begin{itemize}
            \item \textbf{Tensione}: 51.2 V nominali.
            \item \textbf{Capacità}: 80 Ah.
            \item \textbf{Energia Totale}: $51.2 \text{ V} \times 80 \text{ Ah} \approx \textbf{4.1 kWh}$.
        \end{itemize}
        \vspace{0.3cm}
        \textbf{Verifica Autonomia}:
        $$T_{op} = \frac{E_{batt} \cdot \eta_{sys}}{P_{avg}} \approx \frac{4.1 \text{ kWh} \cdot 0.9}{0.8 \text{ kW}} > 4 \text{ h}$$
        \textit{(Considerando consumo medio inferiore al picco)}

% Slide 4: Selezione Motori
\begin{frame}
    \frametitle{Selezione Motori: CFR MRT05}
    Configurazione: \textbf{2 Ruote Motrici Sterzanti} (Motorwheels) + 2 Caster.

    \begin{columns}
        \column{0.4\textwidth}
        \begin{itemize}
            \item \textbf{Trazione}:
                  \begin{itemize}
                      \item Potenza: 500 W (S2-60min).
                      \item Coppia Nominale: 37 Nm.
                      \item Coppia Picco: 180 Nm.
                  \end{itemize}
            \item \textbf{Sterzo}:
                  \begin{itemize}
                      \item Potenza: 300 W.
                      \item Coppia: 35 Nm.
                  \end{itemize}
        \end{itemize}

        \column{0.5\textwidth}
        \begin{block}{Verifica Coppia}
            Coppia alla ruota richiesta in salita ($r=0.15$ m):
            $$C_{req} = F_{salita} \cdot r / 2 \approx 1078 \cdot 0.15 / 2 \approx 81 \text{ Nm}$$
            $$81 \text{ Nm} < 180 \text{ Nm (Picco)} \implies \textbf{OK}$$
        \end{block}
    \end{columns}
\end{frame}
